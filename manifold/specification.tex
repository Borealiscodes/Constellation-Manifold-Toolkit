\section{Constellation Runtime Model}

\subsection{Constellation Layers}
Let $\mathcal{C} = \{\text{Node}, \text{Surface}, \text{Record},
\text{Flux}, \text{Ternary}, \text{Manifold}\}$ denote the ordered
constellation layer set.

\subsection{Embedding State}
An \textbf{embedding state} is a triple:
\[
E = (v, z, M)
\]
where:
\begin{itemize}
  \item $v$ is a ternary vector in $\{-1,0,+1\}^n$,
  \item $z$ is a geometric coordinate embedding,
  \item $M$ is a membrane envelope enforcing bounds and invariants.
\end{itemize}

\subsection{Transitions}
A transition is a function:
\[
T : E \rightarrow E'
\]
subject to membrane and layer constraints.

\section{Operators}

\subsection{Activation Operator}
\[
O_{\text{act}} : \text{Symbolic} \rightarrow \text{Ternary}
\]
parses symbolic telemetry into numeric ternary vectors.

\subsection{Transition Operator}
\[
O_{\text{trans}} : \text{Ternary} \rightarrow \text{Manifold}
\]
embeds ternary vectors into geometric coordinates.

\subsection{Diagnostic Operator}
\[
O_{\text{diag}} : E \rightarrow E
\]
computes drift, roughness, holonomy, and collapse gradients.

\subsection{Reversal Operator}
\[
O_{\text{rev}} : E' \rightarrow E
\]
restores a previous embedding state.

\subsection{Guard Operator}
\[
O_{\text{guard}} : E \rightarrow E
\]
prevents membrane violation or illegal ternary states.

\section{Membrane Constraints}

Allowed transitions:
\[
\text{Symbolic} \rightarrow \text{Ternary} \rightarrow
\text{Manifold} \rightarrow \text{Diagnostic} \rightarrow
\text{Provenance}
\]

Forbidden transitions:
\begin{itemize}
  \item skipping Symbolic $\rightarrow$ Manifold,
  \item skipping Ternary $\rightarrow$ Diagnostic,
  \item irreversible embedding without $O_{\text{rev}}$,
  \item membrane overflow ($v \notin \{-1,0,+1\}$).
\end{itemize}

\section{Reversibility}

A transition $T$ is reversible if:
\[
\exists O_{\text{rev}} : T(E) \rightarrow E
\]

Reversal must preserve:
\begin{itemize}
  \item membrane envelope semantics,
  \item constellation layer ordering,
  \item contamination-free behavior,
  \item ternary integrity.
\end{itemize}
