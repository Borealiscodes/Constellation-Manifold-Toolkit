\section{Constitutional Compatibility Theorem}
\label{sec:compatibility-theorem}

This module establishes the formal compatibility theorem governing interactions
between the NDH Constitutional Ontology and external epistemic or sequencing
frameworks, including the Sovereign-Library ontology and the Dignity Omnibus
Sequencing Document v1.0. The theorem proves non-absorption, identity
preservation, curvature neutrality, lineage reversibility, and reconstruction
safety across all cross-system adjacencies.

% ------------------------------------------------------------
% 1. Preliminaries
% ------------------------------------------------------------

\subsection{Preliminaries}

Let $\mathcal{M}$ be the NDH constitutional manifold with identity fibration
\[
\pi : \mathcal{M} \to I,
\]
curvature field $\kappa$, preferred direction field $\vec{d}$, and
reconstruction sheaf $\mathcal{Q}$.

Let $\mathfrak{E}$ denote any external epistemic or sequencing system, such as
the Sovereign-Library ontology or the Dignity Omnibus sequencing spine.

Let $\Lambda = (\lambda_{\mathrm{out}}, \lambda_{\mathrm{in}})$ denote a
cross-system lineage channel.

% ------------------------------------------------------------
% 2. Compatibility Conditions
% ------------------------------------------------------------

\subsection{Compatibility Conditions}

Cross-system compatibility requires:

\paragraph{Identity Preservation}
\[
\pi \circ \lambda_{\mathrm{in}} = \pi.
\]

\paragraph{Curvature Neutrality}
\[
\kappa(\lambda_{\mathrm{out}}(x)) \le \kappa_{\max}.
\]

\paragraph{Direction Preservation}
\[
\lambda_{\mathrm{out}}(\vec{d}) = \vec{d}.
\]

\paragraph{Lineage Reversibility}
\[
\lambda_{\mathrm{in}} \circ \lambda_{\mathrm{out}} = \mathrm{id}_{\mathcal{M}}.
\]

\paragraph{Reconstruction Safety}
\[
\lambda_{\mathrm{out}}(\mathcal{Q}) = \mathcal{Q}.
\]

% ------------------------------------------------------------
% 3. External System Constraints
% ------------------------------------------------------------

\subsection{External System Constraints}

External systems must satisfy:

\paragraph{Non-Absorption}
\[
\mathrm{Absorp}_{\mathfrak{E}}(x_1, x_2) = \varnothing.
\]

\paragraph{Epistemic Invariance}
\[
\mathfrak{E}(x_1) = \mathfrak{E}(x_1).
\]

\paragraph{Continuity Envelope}
\[
\mathrm{ContEnv}(x) \le \mathrm{Cont}_{\max}.
\]

\paragraph{Reversible Routing}
\[
\lambda_{\mathrm{in}} \circ \lambda_{\mathrm{out}} = \mathrm{id}.
\]

These constraints hold for both the Sovereign-Library ontology and the Dignity
Omnibus sequencing spine.

% ------------------------------------------------------------
% 4. Compatibility Theorem
% ------------------------------------------------------------

\subsection{Compatibility Theorem}

\begin{theorem}[Constitutional Compatibility]
Let $\mathcal{M}$ be the NDH constitutional manifold and let $\mathfrak{E}$ be
any external epistemic or sequencing system satisfying the constraints above.
Then all cross-system adjacencies are constitutionally compatible, and
cross-system absorption is impossible:

\[
\mathrm{Absorp}_{NDH \times \mathfrak{E}} = \varnothing.
\]

Furthermore, all apex invariants are preserved:

\[
\mathfrak{I}_{\mathrm{apex}}(\lambda_{\mathrm{out}}(x))
=
\mathfrak{I}_{\mathrm{apex}}(x).
\]
\end{theorem}

% ------------------------------------------------------------
% 5. Proof
% ------------------------------------------------------------

\subsection{Proof}

\paragraph{Identity Preservation}
Since $\pi \circ \lambda_{\mathrm{in}} = \pi$, identity fibration is preserved
under cross-system routing.

\paragraph{Curvature Neutrality}
Since $\kappa(\lambda_{\mathrm{out}}(x)) \le \kappa_{\max}$, curvature remains
within constitutional bounds.

\paragraph{Direction Preservation}
Since $\lambda_{\mathrm{out}}(\vec{d}) = \vec{d}$, preferred direction is
invariant under cross-system adjacency.

\paragraph{Lineage Reversibility}
Since $\lambda_{\mathrm{in}} \circ \lambda_{\mathrm{out}} = \mathrm{id}$,
lineage channels are reversible and cannot absorb NDH structure.

\paragraph{Reconstruction Safety}
Since $\lambda_{\mathrm{out}}(\mathcal{Q}) = \mathcal{Q}$, reconstruction
behavior is preserved.

\paragraph{Non-Absorption}
Since $\mathrm{Absorp}_{\mathfrak{E}} = \varnothing$, external systems cannot
absorb NDH identity, curvature, direction, lineage, or reconstruction
structures.

Thus:

\[
\mathrm{Absorp}_{NDH \times \mathfrak{E}} = \varnothing.
\]

All apex invariants remain preserved, completing the proof.
