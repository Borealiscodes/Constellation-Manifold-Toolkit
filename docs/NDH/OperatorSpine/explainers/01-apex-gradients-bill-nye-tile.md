\[
\nabla_{\text{apex}} \mathcal{O}(x)
= \operatorname*{arg\,max}_{\|v\|=1}
\left\langle \nabla \mathcal{O}(x), v \right\rangle.

\[
\nabla_{\text{apex}} \mathcal{O}(x)
